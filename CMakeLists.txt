cmake_minimum_required(VERSION 3.16)
project(NakiLiveTranscode VERSION 0.4.0 LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Only Linux / WSL is supported")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavfilter libavutil libswscale)
pkg_check_modules(EVENT REQUIRED IMPORTED_TARGET libevent>=2.1.12)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl>=7.85)
pkg_check_modules(UNWIND REQUIRED IMPORTED_TARGET libunwind-ptrace)
add_library(yyjson STATIC third_party/yyjson/src/yyjson.c)
target_include_directories(yyjson PUBLIC third_party/yyjson/src)
add_executable(naki-live app/service.c app/ipc.c app/common.c app/logger.c app/netstats.c app/crash_client.c)
add_executable(naki-live-worker app/worker.c app/ipc.c app/pipeline.c app/common.c app/logger.c app/netstats.c app/crash_client.c)
foreach(target naki-live naki-live-worker)
    target_compile_definitions(${target} PRIVATE _POSIX_C_SOURCE=200809L)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror=implicit-function-declaration)
    target_link_options(${target} PRIVATE -Wl,-z,now)
    target_link_libraries(${target} PRIVATE yyjson PkgConfig::FFMPEG Threads::Threads m)
    if(NAKI_FFMPEG_PREFIX)
        set_target_properties(${target} PROPERTIES BUILD_RPATH "${NAKI_FFMPEG_PREFIX}/lib" INSTALL_RPATH "${NAKI_FFMPEG_PREFIX}/lib")
    endif()
endforeach()
target_link_libraries(naki-live PRIVATE PkgConfig::EVENT)
add_executable(naki-crash-collector app/crash_collector.c app/crash_client.c)
target_compile_options(naki-crash-collector PRIVATE -Wall -Wextra -Wpedantic -Werror=implicit-function-declaration)
target_link_options(naki-crash-collector PRIVATE -Wl,-z,now)
target_link_libraries(naki-crash-collector PRIVATE yyjson PkgConfig::UNWIND Threads::Threads)
add_executable(naki-live-ctl app/client.c)
target_compile_options(naki-live-ctl PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(naki-live-ctl PRIVATE PkgConfig::CURL)
include(CTest)
if(BUILD_TESTING)
    add_executable(naki-crash-fixture tests/crash_fixture.c app/crash_client.c)
    target_include_directories(naki-crash-fixture PRIVATE app)
    target_compile_options(naki-crash-fixture PRIVATE -g -O1 -fno-omit-frame-pointer -fno-optimize-sibling-calls)
    target_link_options(naki-crash-fixture PRIVATE -Wl,--export-dynamic -Wl,-z,now)
    target_link_libraries(naki-crash-fixture PRIVATE Threads::Threads)
    add_library(naki-unwind-stall SHARED tests/crash_unwind_stall.c)
    target_link_libraries(naki-unwind-stall PRIVATE PkgConfig::UNWIND)
    set_target_properties(naki-unwind-stall PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
    add_executable(naki-logger-test tests/logger_test.c app/logger.c app/crash_client.c)
    target_include_directories(naki-logger-test PRIVATE app)
    target_compile_definitions(naki-logger-test PRIVATE _POSIX_C_SOURCE=200809L)
    target_compile_options(naki-logger-test PRIVATE -UNDEBUG)
    target_link_libraries(naki-logger-test PRIVATE yyjson Threads::Threads)
    add_executable(naki-network-probe tests/network_probe.c)
    target_link_libraries(naki-network-probe PRIVATE PkgConfig::FFMPEG)
    set_target_properties(naki-network-probe PROPERTIES BUILD_RPATH "${NAKI_FFMPEG_PREFIX}/lib")
endif()
