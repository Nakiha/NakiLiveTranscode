function(naki_configure_service target)
    target_include_directories(${target} PRIVATE app)
    target_compile_definitions(${target} PRIVATE _POSIX_C_SOURCE=200809L)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror=implicit-function-declaration)
    target_link_options(${target} PRIVATE -Wl,-z,now)
    target_link_libraries(${target} PRIVATE naki-control yyjson PkgConfig::FFMPEG Threads::Threads m)
    if(NAKI_FFMPEG_PREFIX)
        set_target_properties(${target} PROPERTIES BUILD_RPATH "${NAKI_FFMPEG_PREFIX}/lib" INSTALL_RPATH "${NAKI_FFMPEG_PREFIX}/lib")
    endif()
endfunction()
foreach(target naki-live naki-live-worker)
    naki_configure_service(${target})
endforeach()
