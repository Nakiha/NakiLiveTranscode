cmake_minimum_required(VERSION 3.16)
project(NakiLiveTranscode VERSION 0.1.0 LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Only Linux / WSL is supported")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavfilter libavutil libswscale)
pkg_check_modules(EVENT REQUIRED IMPORTED_TARGET libevent>=2.1.12)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl>=7.85)
add_library(yyjson STATIC third_party/yyjson/src/yyjson.c)
target_include_directories(yyjson PUBLIC third_party/yyjson/src)
add_executable(naki-live app/service.c app/pipeline.c app/common.c)
target_compile_definitions(naki-live PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(naki-live PRIVATE -Wall -Wextra -Wpedantic -Werror=implicit-function-declaration)
target_link_libraries(naki-live PRIVATE yyjson PkgConfig::FFMPEG PkgConfig::EVENT Threads::Threads m)
if(NAKI_FFMPEG_PREFIX)
    set_target_properties(naki-live PROPERTIES BUILD_RPATH "${NAKI_FFMPEG_PREFIX}/lib" INSTALL_RPATH "${NAKI_FFMPEG_PREFIX}/lib")
endif()
add_executable(naki-live-ctl app/client.c)
target_compile_options(naki-live-ctl PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(naki-live-ctl PRIVATE PkgConfig::CURL)
